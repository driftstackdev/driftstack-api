import { Driftstack } from '@driftstack/sdk';

const ds = new Driftstack({ apiKey: process.env.DRIFTSTACK_API_KEY });

const session = await ds.sessions.create({
  archetype: 'iphone17_ios18_7_safari26_4',
  label: 'target-flow',
});

try {
  await ds.sessions.navigate(session.id, { url: 'https://target.example' });
  await ds.sessions.capture(session.id, { kind: 'screenshot' });
  const state = await ds.sessions.getState(session.id);
  console.log(state.url, state.title);
} finally {
  await ds.sessions.destroy(session.id);
}

How it works

Three steps to a real iPhone in the cloud.

1

Pick an iPhone profile

Choose a device archetype — a complete, coherent iPhone identity (hardware, iOS, Safari, fonts, canvas, TLS) that stays consistent for the profile's lifetime.

2

Start a session

One API call or one click in the desktop app boots a real WebKit browser in the cloud, wearing that identity, routed through your proxy.

3

Drive it your way

Watch and tap it live over WebRTC, script it from TypeScript / Python / Go, or hand the task to the AI agent with your own Anthropic key.

Privacy

Your data is yours. Cryptographically.
